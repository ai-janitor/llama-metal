set(METAL_KARGS_DEPS "")
set(METAL_KARGS_SED "")
foreach(h ${METAL_KARGS_HEADERS})
    list(APPEND METAL_KARGS_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/${h}")
    # the path separator is matched by "."
    string(REPLACE "/" "." h_re "${h}")
    list(APPEND METAL_KARGS_SED -e "/\#include \"${h_re}\"/r ${CMAKE_CURRENT_SOURCE_DIR}/${h}" -e "/\#include \"${h_re}\"/d")
endforeach()

if (GGML_METAL_EMBED_LIBRARY)
    enable_language(ASM)

    add_compile_definitions(GGML_METAL_EMBED_LIBRARY)

    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")

    set(METALLIB_EMBED_ASM    "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-embed.s")
    set(METALLIB_SOURCE_EMBED "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-embed.metal")

    # Collect absolute paths for DEPENDS
    set(METAL_KERNEL_DEPS "")
    foreach(src ${METAL_KERNEL_SOURCES})
        list(APPEND METAL_KERNEL_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/${src}")
    endforeach()

    # Concatenate all kernel files into a single .metal blob for embedding:
    #   1. Cat 00-common.metal as-is (has the __embed_ggml-common.h__ marker)
    #   2. Cat remaining kernel files, stripping their #include "00-common.metal" lines
    #   3. Inline ggml-common.h at the __embed marker
    #   4. Inline ggml-metal-impl.h at its #include, then the family headers it includes
    #   5. Wrap in .s assembly for .incbin embedding
    add_custom_command(
        OUTPUT "${METALLIB_EMBED_ASM}"
        # P11.66: the assembled MSL is a side product of this rule; declaring it lets the
        # opt-in ggml-metal-test-metallib target depend on it (generated-file dependency edge).
        BYPRODUCTS "${METALLIB_SOURCE_EMBED}"
        COMMAND echo "Embedding Metal library from kernels/"
        # Step 1+2: concatenate kernel files, strip internal shared-module includes
        COMMAND cat ${METAL_KERNEL_DEPS} | grep -v "^#include \"00-common.metal\"\\|^#include \"dequant.metal\"\\|^#include \"lane-fma-f32.metal\"\\|^#include \"block-dot-basic.metal\"\\|^#include \"block-dot-q2-k.metal\"\\|^#include \"block-dot-q3-k.metal\"\\|^#include \"block-dot-q4-k.metal\"\\|^#include \"block-dot-q8-0.metal\"\\|^#include \"block-dot.metal\"\\|^#include \"block-dot-q6-grouped.metal\"" > "${METALLIB_SOURCE_EMBED}.tmp"
        # Step 3: inline ggml-common.h
        COMMAND sed -e "/__embed_ggml-common.h__/r ${METALLIB_COMMON}" -e "/__embed_ggml-common.h__/d" < "${METALLIB_SOURCE_EMBED}.tmp" > "${METALLIB_SOURCE_EMBED}.tmp2"
        # Step 4: inline ggml-metal-impl.h
        COMMAND sed -e "/\#include \"ggml-metal-impl.h\"/r ${METALLIB_IMPL}" -e "/\#include \"ggml-metal-impl.h\"/d" < "${METALLIB_SOURCE_EMBED}.tmp2" > "${METALLIB_SOURCE_EMBED}.tmp3"
        COMMAND sed ${METAL_KARGS_SED} < "${METALLIB_SOURCE_EMBED}.tmp3" > "${METALLIB_SOURCE_EMBED}.tmp4"
        # Step 5: inline tiled kernel loader includes
        COMMAND sed -e "/\#include \"kernel-tiled-loaders-qk32.metal\"/r ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-qk32.metal" -e "/\#include \"kernel-tiled-loaders-qk32.metal\"/d" -e "/\#include \"kernel-tiled-loaders-kquant.metal\"/r ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-kquant.metal" -e "/\#include \"kernel-tiled-loaders-kquant.metal\"/d" -e "/\#include \"kernel-tiled-loaders-iq.metal\"/r ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-iq.metal" -e "/\#include \"kernel-tiled-loaders-iq.metal\"/d" < "${METALLIB_SOURCE_EMBED}.tmp4" > "${METALLIB_SOURCE_EMBED}"
        # Step 5: generate .s assembly
        COMMAND echo ".section __DATA,__ggml_metallib"          >  "${METALLIB_EMBED_ASM}"
        COMMAND echo ".globl _ggml_metallib_start"              >> "${METALLIB_EMBED_ASM}"
        COMMAND echo "_ggml_metallib_start:"                    >> "${METALLIB_EMBED_ASM}"
        COMMAND echo .incbin "\"${METALLIB_SOURCE_EMBED}\""     >> "${METALLIB_EMBED_ASM}"
        COMMAND echo ".globl _ggml_metallib_end"                >> "${METALLIB_EMBED_ASM}"
        COMMAND echo "_ggml_metallib_end:"                      >> "${METALLIB_EMBED_ASM}"
        # Cleanup temp files
        COMMAND rm -f "${METALLIB_SOURCE_EMBED}.tmp" "${METALLIB_SOURCE_EMBED}.tmp2" "${METALLIB_SOURCE_EMBED}.tmp3" "${METALLIB_SOURCE_EMBED}.tmp4"
        DEPENDS ${METAL_KERNEL_DEPS} ${METALLIB_COMMON} ${METALLIB_IMPL} ${METAL_KARGS_DEPS}
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-qk32.metal
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-kquant.metal
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-iq.metal
        COMMENT "Generate assembly for embedded Metal library"
        VERBATIM
    )

    target_sources(ggml-metal PRIVATE "${METALLIB_EMBED_ASM}")

    # P11.66: opt-in offline payload (not in ALL). Compiles the embedded MSL (the same text
    # embedded in the production binary, included once) plus test-contract-probe.metal into
    # bin/ggml-metal-test.metallib without loading a device. Flags mirror the non-embedded path.
    set(METALLIB_PROBE   "${CMAKE_CURRENT_SOURCE_DIR}/test-contract-probe.metal")
    set(METALLIB_TEST_WRAPPER "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-test-wrapper.metal")
    set(METALLIB_TEST_AIR     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-test.air")
    set(METALLIB_TEST_OUT     "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal-test.metallib")

    if (GGML_METAL_SHADER_DEBUG)
        set(TEST_XC_FLAGS -fno-fast-math -fno-inline)
    else()
        set(TEST_XC_FLAGS -O3)
    endif()
    if (GGML_METAL_MACOSX_VERSION_MIN)
        list(APPEND TEST_XC_FLAGS -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN})
    endif()
    if (GGML_METAL_STD)
        list(APPEND TEST_XC_FLAGS -std=${GGML_METAL_STD})
    endif()

    # Wrapper includes each source exactly once (an include, not a second concatenation).
    file(WRITE "${METALLIB_TEST_WRAPPER}"
"// generated by ggml/src/ggml-metal/CMakeLists.txt (P11.66); do not edit
#include \"${METALLIB_SOURCE_EMBED}\"
#include \"${METALLIB_PROBE}\"
")

    add_custom_command(
        OUTPUT "${METALLIB_TEST_OUT}"
        # -DGGML_METAL_EMBED_LIBRARY=1: same macro ggml-metal-device.m passes the runtime compile (guards the embedded text)
        COMMAND xcrun -sdk macosx metal ${TEST_XC_FLAGS} -DGGML_METAL_EMBED_LIBRARY=1 -c "${METALLIB_TEST_WRAPPER}" -o "${METALLIB_TEST_AIR}"
        COMMAND xcrun -sdk macosx metallib "${METALLIB_TEST_AIR}" -o "${METALLIB_TEST_OUT}"
        DEPENDS "${METALLIB_EMBED_ASM}" "${METALLIB_SOURCE_EMBED}" "${METALLIB_PROBE}" "${METALLIB_TEST_WRAPPER}"
                ${METAL_KERNEL_DEPS} ${METALLIB_COMMON} ${METALLIB_IMPL} ${METAL_KARGS_DEPS}
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-qk32.metal
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-kquant.metal
                ${CMAKE_CURRENT_SOURCE_DIR}/matmul/kernel-tiled-loaders-iq.metal
        COMMENT "Compiling offline Metal test payload"
        VERBATIM
    )
    # SOURCES: metadata only (no compile); lets the File-API source collection see the probe this target owns.
    add_custom_target(ggml-metal-test-metallib DEPENDS "${METALLIB_TEST_OUT}" SOURCES "${METALLIB_PROBE}")
else()
    # Non-embedded path: copy kernel files to bin directory, compile to metallib
    # Copy headers needed by Metal compiler
    configure_file(../ggml-common.h  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-common.h  COPYONLY)
    configure_file(ggml-metal-impl.h ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal-impl.h COPYONLY)
    foreach(h ${METAL_KARGS_HEADERS})
        configure_file(${h} ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${h} COPYONLY)
    endforeach()

    # Copy all kernel files to bin directory
    foreach(src ${METAL_KERNEL_SOURCES})
        get_filename_component(fname ${src} NAME)
        configure_file(${src} ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${fname} COPYONLY)
    endforeach()

    # Copy tiled loader include files (not in METAL_KERNEL_SOURCES, inlined via #include)
    configure_file(matmul/kernel-tiled-loaders-qk32.metal   ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/kernel-tiled-loaders-qk32.metal   COPYONLY)
    configure_file(matmul/kernel-tiled-loaders-kquant.metal  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/kernel-tiled-loaders-kquant.metal  COPYONLY)
    configure_file(matmul/kernel-tiled-loaders-iq.metal      ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/kernel-tiled-loaders-iq.metal      COPYONLY)

    if (GGML_METAL_SHADER_DEBUG)
        set(XC_FLAGS -fno-fast-math -fno-inline)
    else()
        set(XC_FLAGS -O3)
    endif()

    if (GGML_METAL_MACOSX_VERSION_MIN)
        message(STATUS "Adding  -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN} flag to metal compilation")
        list   (APPEND XC_FLAGS -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN})
    endif()

    if (GGML_METAL_STD)
        message(STATUS "Adding  -std=${GGML_METAL_STD} flag to metal compilation")
        list   (APPEND XC_FLAGS -std=${GGML_METAL_STD})
    endif()

    # Concatenate all kernel files into one .metal, strip internal includes, compile
    set(METALLIB_MERGED "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal-merged.metal")

    # Collect bin-dir kernel paths for the compile command
    set(METAL_KERNEL_BIN_DEPS "")
    foreach(src ${METAL_KERNEL_SOURCES})
        get_filename_component(fname ${src} NAME)
        list(APPEND METAL_KERNEL_BIN_DEPS "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${fname}")
    endforeach()

    add_custom_command(
        OUTPUT ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        COMMAND cat ${METAL_KERNEL_BIN_DEPS} | grep -v "^#include \"00-common.metal\"\\|^#include \"dequant.metal\"\\|^#include \"lane-fma-f32.metal\"\\|^#include \"block-dot-basic.metal\"\\|^#include \"block-dot-q2-k.metal\"\\|^#include \"block-dot-q3-k.metal\"\\|^#include \"block-dot-q4-k.metal\"\\|^#include \"block-dot-q8-0.metal\"\\|^#include \"block-dot.metal\"\\|^#include \"block-dot-q6-grouped.metal\"" > "${METALLIB_MERGED}"
        COMMAND xcrun -sdk macosx metal ${XC_FLAGS}
                -I ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}
                -c "${METALLIB_MERGED}" -o - |
                xcrun -sdk macosx metallib - -o ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        COMMAND rm -f ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-common.h
        COMMAND rm -f "${METALLIB_MERGED}"
        DEPENDS ${METAL_KERNEL_DEPS} ${METALLIB_COMMON}
        COMMENT "Compiling Metal kernels"
        )

    add_custom_target(
        ggml-metal-lib ALL
        DEPENDS ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        )
endif() # GGML_METAL_EMBED_LIBRARY
